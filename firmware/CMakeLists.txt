cmake_minimum_required(VERSION 3.20)
project(car_sniffer C ASM)
set(CMAKE_C_STANDARD 11)
set(CUBE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/STM32CubeF4)
set(HAL ${CUBE}/Drivers/STM32F4xx_HAL_Driver)
set(CMSIS ${CUBE}/Drivers/CMSIS)
set(USB ${CUBE}/Middlewares/ST/STM32_USB_Device_Library)
if(NOT EXISTS ${HAL}/Src/stm32f4xx_hal.c)
  message(FATAL_ERROR "Run scripts/bootstrap.ps1 first")
endif()
set(HAL_MODULES hal hal_rcc hal_rcc_ex hal_gpio hal_cortex hal_dma hal_flash hal_flash_ex hal_pwr hal_pwr_ex hal_can hal_adc hal_adc_ex hal_tim hal_tim_ex hal_pcd hal_pcd_ex hal_rng ll_usb)
foreach(M IN LISTS HAL_MODULES)
  list(APPEND HAL_SOURCES ${HAL}/Src/stm32f4xx_${M}.c)
endforeach()
add_executable(car_sniffer.elf src/main.c src/engine.c src/protocol.c src/usb_transport.c ${HAL_SOURCES}
 ${CMSIS}/Device/ST/STM32F4xx/Source/Templates/system_stm32f4xx.c
 ${CMSIS}/Device/ST/STM32F4xx/Source/Templates/gcc/startup_stm32f407xx.s
 ${USB}/Core/Src/usbd_core.c ${USB}/Core/Src/usbd_ctlreq.c ${USB}/Core/Src/usbd_ioreq.c ${USB}/Class/CDC/Src/usbd_cdc.c)
target_include_directories(car_sniffer.elf PRIVATE include ${HAL}/Inc ${CMSIS}/Include ${CMSIS}/Device/ST/STM32F4xx/Include ${USB}/Core/Inc ${USB}/Class/CDC/Inc)
target_compile_definitions(car_sniffer.elf PRIVATE STM32F407xx USE_HAL_DRIVER HSE_VALUE=8000000U)
target_compile_options(car_sniffer.elf PRIVATE -mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=hard -ffunction-sections -fdata-sections -Og -g3 -Wall -Wextra)
target_link_options(car_sniffer.elf PRIVATE -mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=hard -T${CMAKE_CURRENT_SOURCE_DIR}/STM32F407VE.ld --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/car_sniffer.map -Wl,--print-memory-usage)
target_link_libraries(car_sniffer.elf PRIVATE m)
add_custom_command(TARGET car_sniffer.elf POST_BUILD COMMAND arm-none-eabi-objcopy -O binary $<TARGET_FILE:car_sniffer.elf> ${CMAKE_CURRENT_BINARY_DIR}/car_sniffer.bin COMMAND arm-none-eabi-size $<TARGET_FILE:car_sniffer.elf>)
